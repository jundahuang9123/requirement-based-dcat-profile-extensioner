% Fig. 1: the pipeline. Mirrors the RQ1 paper's Fig. 1 (numbered operations in three labelled rows,
% the middle row read from right to left). Shaded: the operation calls the model; double border:
% a person decides; plain: program code.
\begin{tikzpicture}[
    font=\scriptsize,
    op/.style={draw, rounded corners=2pt, minimum height=0.9cm, align=center,
               inner sep=2pt, text width=3.5cm},
    llm/.style={op, fill=black!12},
    human/.style={op, double, double distance=0.9pt},
    row/.style={font=\scriptsize\bfseries, anchor=south west, inner sep=1pt},
    flow/.style={-{Stealth[length=1.5mm]}, semithick},
    side/.style={-{Stealth[length=1.5mm]}, semithick, dashed},
  ]
  % row 1: read and propose (left to right)
  \node[op]  (o1) at (0,0)    {1.\ Intake: lint the atoms,\\ map subjects to classes};
  \node[op]  (o2) at (4.1,0)  {2.\ Retrieve candidates, base\\ elements of the class first};
  \node[llm] (o3) at (8.2,0)  {3.\ Propose a disposition and\\ typed changes per atom};
  % row 2: check and repair (right to left)
  \node[llm] (o4) at (8.2,-1.6) {4.\ Verify: V1--V6, T1, B1 in\\ code, T2 by a model judge};
  \node[llm] (o5) at (4.1,-1.6) {5.\ Repair failing changes,\\ at most three rounds};
  \node[op]  (o6) at (0,-1.6)   {6.\ Consolidate: one element\\ per class and property};
  % row 3: compile and release (left to right)
  \node[op]    (o7) at (0,-3.2)   {7.\ Compile SHACL, LinkML,\\ DSV, ProfileDelta, docs};
  \node[op]    (o8) at (4.1,-3.2) {8.\ Conformance C1--C7,\\ witness vs.\ official base};
  \node[human] (o9) at (8.2,-3.2) {9.\ Review the queue, decide,\\ sign off the release};
  % row labels
  \node[row] at (o1.north west) {Read and propose};
  \node[row] at (o6.north west) {Check and repair};
  \node[row] at (o7.north west) {Compile and release};
  % flow
  \draw[flow] (o1) -- (o2);
  \draw[flow] (o2) -- (o3);
  \draw[flow] (o3) -- (o4);
  \draw[flow] ([yshift=1.5mm]o4.west) -- ([yshift=1.5mm]o5.east);
  \draw[side] ([yshift=-1.5mm]o5.east) -- ([yshift=-1.5mm]o4.west);
  \draw[flow] (o5) -- (o6);
  \draw[flow] ([xshift=12mm]o6.south) -- ([xshift=12mm]o7.north);
  \draw[flow] (o7) -- (o8);
  \draw[flow] (o8) -- (o9);
  \draw[side] (o4.south) -- node[right=1pt, font=\tiny, align=left] {queue: unresolved,\\ unsure, conflicting} (o9.north);
  % blackboard
  \node[draw, dashed, rounded corners=2pt, anchor=north, text width=11.6cm, align=center,
        inner sep=3pt] (bb) at (4.1,-3.9)
        {Blackboard: append-only log of every rule, model call and human action\\
         current state of every atom and change\ $\cdot$\ model calls with prompts, answers, tokens and a replay cache};
\end{tikzpicture}
